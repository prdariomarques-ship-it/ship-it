"""Vigia das linhas do WhatsApp: consulta `GET /instance/connectionState/<instancia>` no Evolution e decide quando avisar o dono.

Motivo: em 16/10/2026 a linha da loja caiu (`device_removed`, estado `connecting`) e ficou ~12 h fora sem ninguem perceber; nenhuma regra do
Prometheus cobria isso. Este modulo e ADITIVO: nao toca no provedor do WhatsApp nem no fluxo das mensagens. So le o estado e, quando
preciso, enfileira um `whatsapp.send_text` para o dono por uma linha que esteja `open` (preferindo uma que nao seja a pessoal).

Regras: so avisa depois de `confirm_checks` rodadas seguidas fora do ar; nao repete antes de `repeat_minutes`; avisa uma vez quando volta.
Se NENHUMA linha esta no ar (ou o Evolution nao responde), nao ha por onde avisar: so registra no log.
"""
from __future__ import annotations

from dataclasses import dataclass, field
from urllib.parse import quote

import httpx

from services.rate_limit import rate_limiter

OPEN = "open"
UNREACHABLE = "unreachable"


@dataclass
class LineEvent:
    kind: str          # "down" | "still_down" | "recovered"
    instance: str
    state: str
    checks_down: int = 0


@dataclass
class LineWatchState:
    bad_checks: dict[str, int] = field(default_factory=dict)
    alerted: set[str] = field(default_factory=set)

    def reset(self) -> None:
        self.bad_checks.clear()
        self.alerted.clear()


line_watch_state = LineWatchState()


def watched_instances(settings) -> list[str]:
    names = [n.strip() for n in settings.line_watch_instances.split(",") if n.strip()]
    if not names:
        names = [n for n in (settings.evolution_instance, settings.evolution_personal_instance) if n]
    return list(dict.fromkeys(names))


async def fetch_states(instances: list[str], *, base_url: str, api_key: str, timeout: float = 15.0,
                       transport: httpx.AsyncBaseTransport | None = None) -> dict[str, str]:
    """Estado de cada instancia segundo o Evolution; erro de rede/HTTP vira `unreachable` (nunca levanta)."""
    states: dict[str, str] = {}
    async with httpx.AsyncClient(timeout=timeout, transport=transport) as client:
        for instance in instances:
            try:
                response = await client.get(f"{base_url.rstrip('/')}/instance/connectionState/{quote(instance, safe='')}",
                                            headers={"apikey": api_key})
                if response.status_code != 200:
                    states[instance] = UNREACHABLE
                    continue
                state = (response.json().get("instance") or {}).get("state")
                states[instance] = str(state) if state else UNREACHABLE
            except (httpx.HTTPError, ValueError, AttributeError):
                states[instance] = UNREACHABLE
    return states


async def evaluate(states: dict[str, str], *, confirm_checks: int, repeat_minutes: int,
                   state: LineWatchState = line_watch_state) -> list[LineEvent]:
    """Atualiza o estado em memoria e devolve os avisos devidos nesta rodada."""
    events: list[LineEvent] = []
    for instance, current in states.items():
        if current == OPEN:
            state.bad_checks[instance] = 0
            if instance in state.alerted:
                state.alerted.discard(instance)
                events.append(LineEvent("recovered", instance, current))
            continue
        bad = state.bad_checks.get(instance, 0) + 1
        state.bad_checks[instance] = bad
        if bad < max(1, confirm_checks):
            continue
        # O limitador (Redis, com fallback local) segura repeticoes mesmo se o backend reiniciar no meio da queda.
        allowed = await rate_limiter.is_allowed(f"line-watch:{instance}", limit=1, window_seconds=max(60, repeat_minutes * 60))
        first = instance not in state.alerted
        state.alerted.add(instance)
        if allowed:
            events.append(LineEvent("down" if first else "still_down", instance, current, bad))
    return events


def pick_sender(states: dict[str, str], down_instance: str, personal_instance: str = "") -> str | None:
    """Uma linha no ar (diferente da que caiu), preferindo uma que NAO seja a pessoal do dono."""
    open_lines = [i for i, s in states.items() if s == OPEN and i != down_instance]
    others = [i for i in open_lines if i != personal_instance]
    return (others or open_lines or [None])[0]


def message_for(event: LineEvent, interval_seconds: int) -> str:
    if event.kind == "recovered":
        return f"✅ A linha {event.instance} voltou (open). Alerta automático, não responda."
    minutes = max(1, round(event.checks_down * interval_seconds / 60))
    again = " (lembrete)" if event.kind == "still_down" else ""
    return (f"⚠️ A linha {event.instance} está FORA DO AR{again}: estado \"{event.state}\" há cerca de {minutes} min. "
            "Veja o aparelho (Aparelhos conectados) e o QR no painel do Evolution. Alerta automático, não responda.")

"""Job `lines.watch`: vigia das linhas do WhatsApp, autorreagendavel (mesmo padrao de `observation/scheduler.py` e `jobs/style_job.py`).

Cada rodada consulta o Evolution, decide os avisos (`services/line_watch.py`), enfileira `whatsapp.send_text` para o dono por uma linha
no ar e se reagenda. A rodada NUNCA deixa a corrente morrer: qualquer erro vira log e o proximo job e criado mesmo assim.
Desligado por padrao (`LINE_WATCH_ENABLED=false`): a semente `start` nem cria a corrente.
"""
from sqlalchemy.ext.asyncio import AsyncSession

from jobs.registry import job_handler
from jobs.service import JobService
from repositories.job import JobRepository
from services import line_watch
from services.audit import record_log
from utils.config import get_settings
from utils.logging import get_logger

logger = get_logger(__name__)
JOB_NAME = "lines.watch"


async def _run_round(db: AsyncSession, settings, jobs: JobService) -> None:
    instances = line_watch.watched_instances(settings)
    if not instances:
        return
    states = await line_watch.fetch_states(instances, base_url=settings.evolution_base_url, api_key=settings.evolution_api_key)
    events = await line_watch.evaluate(states, confirm_checks=settings.line_watch_confirm_checks,
                                       repeat_minutes=settings.line_watch_repeat_minutes)
    phone = settings.line_watch_alert_phone or settings.whatsapp_owner_alert_phone
    for event in events:
        text = line_watch.message_for(event, settings.line_watch_interval_seconds)
        sender = line_watch.pick_sender(states, event.instance, settings.evolution_personal_instance)
        sent = bool(phone and sender)
        await record_log(
            db, source="line_watch", level="info" if event.kind == "recovered" else "error",
            message=text, payload={"instance": event.instance, "state": event.state, "kind": event.kind,
                                   "sent_via": sender if sent else None, "no_phone": not phone, "no_open_line": sender is None},
        )
        if sent:
            await jobs.enqueue("whatsapp.send_text", {"to": phone, "content": text, "instance": sender})


@job_handler(JOB_NAME)
async def line_watch_tick(db: AsyncSession, payload: dict) -> None:
    settings = get_settings()
    jobs = JobService(db)
    try:
        await _run_round(db, settings, jobs)
    except Exception:  # noqa: BLE001 - a corrente nao pode morrer por um erro numa rodada
        logger.exception("line_watch: erro na rodada; segue para a proxima")
    await jobs.enqueue(JOB_NAME, delay_seconds=settings.line_watch_interval_seconds)


async def start(db: AsyncSession) -> None:
    """Semeia a primeira rodada. Idempotente: nao cria uma segunda corrente."""
    if not get_settings().line_watch_enabled:
        return
    if await JobRepository(db).pending_by_name(JOB_NAME) is not None:
        return
    await JobService(db).enqueue(JOB_NAME, delay_seconds=0)

Em produção o vigia nasce LIGADO (line_watch_enabled=True) com as instâncias e o telefone informados; no repositório o padrão é desligado.
Só o import do job entra no main.py (registra o handler); a primeira rodada é semeada pelo script de aplicação.

import os
import sys
from pathlib import Path

CONFIG_ANCHOR = "    stt_background_enabled: bool = False\n"
CONFIG_ADD = '''    # Vigia das linhas do WhatsApp (services/line_watch.py): consulta o estado de cada instancia no Evolution a cada rodada e avisa o
    # dono quando uma cair (estado diferente de "open"), por OUTRA linha que esteja no ar. Desligado por padrao.
    line_watch_enabled: bool = {enabled}
    # Instancias vigiadas, separadas por virgula. Vazio = a da loja e a pessoal configuradas.
    line_watch_instances: str = {instances}
    line_watch_interval_seconds: int = 120
    line_watch_confirm_checks: int = 2          # rodadas seguidas fora do ar antes de avisar (evita alarme falso)
    line_watch_repeat_minutes: int = 60         # lembrete enquanto a linha continuar fora
    # Telefone (com DDI) que recebe o alerta; vazio = usa whatsapp_owner_alert_phone.
    line_watch_alert_phone: str = {phone}
'''
MAIN_ANCHOR = "import observation.scheduler  # noqa: F401 - register the observation.tick job handler\n"
MAIN_ADD = "import jobs.line_watch_job  # noqa: F401 - register the lines.watch job handler\n"


def once(text: str, anchor: str, what: str) -> None:
    n = text.count(anchor)
    if n != 1:
        raise SystemExit(f"ABORTAR: {what}: âncora encontrada {n} vez(es), esperado 1 (o arquivo do container difere do esperado)")


def build(config: str, main: str, *, enabled=False, instances="", phone="") -> tuple[str, str]:
    if "line_watch" in config or "line_watch" in main:
        raise SystemExit("ABORTAR: parece que o vigia já foi aplicado (não vou aplicar duas vezes)")
    once(config, CONFIG_ANCHOR, "config.py")
    once(main, MAIN_ANCHOR, "main.py")
    add = CONFIG_ADD.replace("{enabled}", repr(bool(enabled))).replace("{instances}", '"' + instances + '"').replace("{phone}", '"' + phone + '"')
    return config.replace(CONFIG_ANCHOR, CONFIG_ANCHOR + add, 1), main.replace(MAIN_ANCHOR, MAIN_ANCHOR + MAIN_ADD, 1)


def main_cli(folder: str) -> None:
    d = Path(folder)
    read = lambda name: (d / f"{name}.lido").read_text(encoding="utf-8")
    instances, phone = os.environ.get("LW_INSTANCES", ""), os.environ.get("LW_PHONE", "")
    if not phone.isdigit() or not 10 <= len(phone) <= 15:
        raise SystemExit("ABORTAR: LW_PHONE deve ter só dígitos (10 a 15, com DDI)")
    if not instances.strip():
        raise SystemExit("ABORTAR: LW_INSTANCES vazio")
    config, main = build(read("config.py"), read("main.py"), enabled=True, instances=instances, phone=phone)
    (d / "config.py.novo").write_text(config, encoding="utf-8", newline="")
    (d / "main.py.novo").write_text(main, encoding="utf-8", newline="")
    print("gerados: config.py.novo main.py.novo")


if __name__ == "__main__":
    if len(sys.argv) != 2:
        raise SystemExit(__doc__)
    main_cli(sys.argv[1])
